Core (Vanilla JS)
import { createMiniApp } from '@webview-sdk/core' const app = createMiniApp({ appId: 'com.demo', debug: true }) const loc = await app.invoke('getLocation') await app.ui.showToast({ title: 'OK' }) app.ready()
React
import { useMiniApp } from '@webview-sdk/react' function App() { const { invoke, ui, ready } = useMiniApp() useEffect(() => ready(), []) return <button onClick={() => invoke('scan')} >Scan</button> }
Vue 3
import { useMiniApp } from '@webview-sdk/vue' const { invoke, ui, ready } = useMiniApp() onMounted(() => ready()) async function scan() { const r = await invoke('scan') await ui.showToast({ title: r.code }) }
